# Notifications

The Notification service is designed to be used alongside a single instance of
the `<uni-notifications>` component, which renders whatever the service emits.
Place one near the root of your layout, then call the service's methods from
anywhere to present the notifications below.

## Example Implementations

### Alert

<Story of={Stories.Alert} />

_Note: The [alert component](?path=/docs/components-feedback-alert--alert) can be implemented independently._

```
import { NotificationService, Alert } from '@uni-design-system/uni-angular';

@Component({...})
class MyComponent {
  constructor(notifications: NotificationService) {}

  const alert: Alert = {

  }

  showAlert = () => this.notifications.showAlert(alert);

}
```

### Snackbar

<Story of={Stories.Snackbar} />

```
import { NotificationService, Snackbar } from '@uni-design-system/uni-angular';

@Component({...})
class MyComponent {
  constructor(notifications: NotificationService) {}

  const snackbar: Snackbar = {
    message: 'The record has been deleted.',
    action?: () => undoDelete();
    actionLabel: 'Undo';
    symbolName: 'delete';
    variant: 'warn';
  }

  showDeleteSnackbar = () => this.notifications.showSnackbar(snackbar);

}
```

### Confirmation

<Story of={Stories.Confirmation} />

```
import { NotificationService, Confirmation } from '@uni-design-system/uni-angular';

@Component({...})
class MyComponent {
  constructor(notifications: NotificationService) {}

  const confirmation: Confirmation = {
      title: 'Delete Record',
      message: 'Are you sure you want to delete this record?',
      action: () => deleteRecord(),
      actionLabel: 'Yes',
      dismissLabel: 'No',
      color: 'warn'
  }

  confirmDelete = () => this.notifications.showConfirmation(confirmation);

}
```
